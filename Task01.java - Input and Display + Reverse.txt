Task01.java - Input and Display + Reverse


import java.util.Scanner; // Import Scanner for user input

public class Task01 { // Class name must match file name
    public static void main(String[] args) { // Main method
        Scanner sc = new Scanner(System.in); // Create Scanner object
        int[] arr = new int[10]; // Declare array of 10 integers
        System.out.println("Enter 10 integers:"); // Ask user for input
        for (int i = 0; i < 10; i++) { // Loop 10 times
            arr[i] = sc.nextInt(); // Store input in array
        }
        System.out.println("Array elements:"); // Heading
        for (int i = 0; i < 10; i++) { // Loop for display
            System.out.print(arr[i] + " "); // Print element
        }
        System.out.println("\nReverse order:"); // Heading for reverse
        for (int i = 9; i >= 0; i--) { // Loop from last to first
            System.out.print(arr[i] + " "); // Print in reverse
        }
        sc.close(); // Close scanner
    }
}
// Time Complexity: O(n)
