import java.util.Scanner;

public class Task10 {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        System.out.println("Enter size of first array:");
        int n1 = sc.nextInt(); // Size 1
        int[] a1 = new int[n1];
        System.out.println("Enter " + n1 + " elements in ascending order:");
        for (int i = 0; i < n1; i++) a1[i] = sc.nextInt();
        System.out.println("Enter size of second array:");
        int n2 = sc.nextInt(); // Size 2
        int[] a2 = new int[n2];
        System.out.println("Enter " + n2 + " elements in ascending order:");
        for (int i = 0; i < n2; i++) a2[i] = sc.nextInt();
        int[] merged = new int[n1 + n2]; // Third array
        for (int i = 0; i < n1; i++) merged[i] = a1[i]; // Copy first
        for (int i = 0; i < n2; i++) merged[n1 + i] = a2[i]; // Copy second
        int max = merged[0]; // Assume first is max
        for (int i = 1; i < merged.length; i++) {
            if (merged[i] > max) max = merged[i]; // Find max
        }
        System.out.println("Merged array:");
        for (int i = 0; i < merged.length; i++) System.out.print(merged[i] + " ");
        System.out.println("\nLargest: " + max);
        sc.close();
    }
}
// Time Complexity: O(n1 + n2)
