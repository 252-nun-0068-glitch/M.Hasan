import java.util.Scanner;

public class Task04 {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int[] arr = new int[10];
        System.out.println("Enter 10 integers:");
        for (int i = 0; i < 10; i++) {
            arr[i] = sc.nextInt();
        }
        System.out.println("Enter number to search:");
        int key = sc.nextInt(); // Number to find
        boolean found = false; // Flag
        System.out.print("Found at indices: ");
        for (int i = 0; i < 10; i++) {
            if (arr[i] == key) { // If match
                System.out.print(i + " "); // Print index
                found = true; // Set flag true
            }
        }
        if (!found) { // If not found
            System.out.println("\nNumber not found."); // Message
        }
        sc.close();
    }
}
// Time Complexity: O(n)
