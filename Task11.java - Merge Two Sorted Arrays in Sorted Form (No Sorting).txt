import java.util.Scanner;

public class Task11 {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        System.out.println("Enter size of first sorted array:");
        int n1 = sc.nextInt();
        int[] a1 = new int[n1];
        System.out.println("Enter " + n1 + " sorted elements:");
        for (int i = 0; i < n1; i++) a1[i] = sc.nextInt();
        System.out.println("Enter size of second sorted array:");
        int n2 = sc.nextInt();
        int[] a2 = new int[n2];
        System.out.println("Enter " + n2 + " sorted elements:");
        for (int i = 0; i < n2; i++) a2[i] = sc.nextInt();
        int[] merged = new int[n1 + n2]; // Result array
        int i = 0, j = 0, k = 0; // Pointers for a1, a2, merged
        while (i < n1 && j < n2) { // While both have elements
            if (a1[i] <= a2[j]) { // If a1 smaller
                merged[k++] = a1[i++]; // Take from a1
            } else {
                merged[k++] = a2[j++]; // Take from a2
            }
        }
        while (i < n1) merged[k++] = a1[i++]; // Copy remaining a1
        while (j < n2) merged[k++] = a2[j++]; // Copy remaining a2
        System.out.println("Merged Sorted array:");
        for (int x = 0; x < merged.length; x++) System.out.print(merged[x] + " ");
        sc.close();
    }
}
// Time Complexity: O(n1 + n2)
