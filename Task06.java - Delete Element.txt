import java.util.Scanner;

public class Task06 {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int[] arr = new int[10];
        System.out.println("Enter 10 integers:");
        for (int i = 0; i < 10; i++) {
            arr[i] = sc.nextInt();
        }
        System.out.println("Enter index (0-9) to delete:");
        int index = sc.nextInt(); // Index to delete
        for (int i = index; i < 9; i++) { // Shift left
            arr[i] = arr[i + 1]; // Move next element
        }
        System.out.println("Updated array:");
        for (int i = 0; i < 9; i++) { // Display 9 elements
            System.out.print(arr[i] + " ");
        }
        sc.close();
    }
}
// Time Complexity: O(n)
