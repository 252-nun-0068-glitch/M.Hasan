import java.util.Scanner;

public class Task05 {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int[] arr = new int[11]; // Size 11 to allow insertion
        System.out.println("Enter 10 integers:");
        for (int i = 0; i < 10; i++) {
            arr[i] = sc.nextInt();
        }
        System.out.println("Enter index (0-10) to insert:");
        int index = sc.nextInt(); // Position
        System.out.println("Enter number to insert:");
        int num = sc.nextInt(); // Value
        for (int i = 10; i > index; i--) { // Shift right
            arr[i] = arr[i - 1]; // Move element
        }
        arr[index] = num; // Insert new element
        System.out.println("Updated array:");
        for (int i = 0; i < 11; i++) {
            System.out.print(arr[i] + " ");
        }
        sc.close();
    }
}
// Time Complexity: O(n)
